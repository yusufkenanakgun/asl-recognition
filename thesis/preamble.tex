% =============================================================================
% Extra package stack layered on TOP of the official Yeditepe template
% (styles/report_format.sty). Compile with XeLaTeX (Times New Roman via
% fontspec). The official .sty already provides: chapter/section heading
% styles, caption/TOC/LOF/LOT formatting, 1.5 line spacing, parskip, the
% cover/approval macros, and the abstract/ozet/acknowledgements environments.
% Here we only add what the thesis body needs, without overriding the
% template's formatting.
% =============================================================================

% --- Math (no math-font package: keep the template's default like the sample) -
\usepackage{amsmath, amsthm, amssymb}

% --- Fonts: real Times New Roman via fontspec (requires XeLaTeX/LuaLaTeX) ------
\RequirePackage{fix-cm}
\usepackage{fontspec}
\setmainfont{Times New Roman}

% --- Bibliography (IEEE numeric, same style the template ships with) ----------
\usepackage[backend=biber, style=ieee, sorting=none, maxbibnames=99]{biblatex}
\addbibresource{references.bib}

% --- Page geometry (official Yeditepe margins) --------------------------------
\usepackage[top=3.5cm, bottom=2cm, left=3.5cm, right=2cm]{geometry}

% --- Header / footer: centered page number at the bottom ----------------------
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0pt}
\fancyfoot[C]{\thepage}

% --- Units & numbers ----------------------------------------------------------
\usepackage{siunitx}
\sisetup{detect-all, group-separator={,}, group-minimum-digits=4}

% --- Graphics & figures -------------------------------------------------------
\usepackage{graphicx}
\graphicspath{{figures/}}
\usepackage{float}
\usepackage{rotating}

% --- Tables -------------------------------------------------------------------
\usepackage{booktabs, array, multirow, tabularx, longtable}

% --- Captions & sub-figures ---------------------------------------------------
% Replicate the template's caption look (bold "Figure N." label, period
% separator, centered) while enabling subfigures used in Chapter 4.
\usepackage{caption}
\usepackage{subcaption}
\captionsetup{font=small, labelfont=bf, labelsep=period, justification=centering}

% --- Algorithms (pseudo-code, Chapter 3) --------------------------------------
\usepackage{algorithm}
\usepackage{algpseudocode}

% --- Misc ---------------------------------------------------------------------
\usepackage{xcolor}
\usepackage{setspace}
\usepackage{microtype}

% Allow a little extra last-resort interword stretch so justified lines that
% end in long, unbreakable monospace tokens (file names, package ids) do not
% spill into the right margin.
\setlength{\emergencystretch}{3em}

% --- Hyperlinks (must load late) ----------------------------------------------
\usepackage[
  colorlinks=true,
  linkcolor=black,
  citecolor=black,
  urlcolor=blue!60!black,
  pdfborder={0 0 0},
  bookmarksnumbered=true,
]{hyperref}
\usepackage[capitalize, noabbrev]{cleveref}

% --- Convenience commands retained from the original draft --------------------
\newcommand{\todo}[1]{\textcolor{red}{\textbf{[TODO: #1]}}}
\newcommand{\thesistitle}{On-Device American Sign Language Recognition: Image- versus Landmark-Based Approaches}
\newcommand{\thesisadvisor}{Betül Arslan}
